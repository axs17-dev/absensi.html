/* ==================================================
   LOGIN
================================================== */

function login() {

    const username =
        document.getElementById(
            "username"
        ).value.trim();

    const password =
        document.getElementById(
            "password"
        ).value;

    const role =
        document.getElementById(
            "role"
        ).value;

    const account =
        accounts[role];


    if (
        username === account.username &&
        password === account.password
    ) {

        localStorage.setItem(
            "loginUser",
            JSON.stringify(account)
        );

        bukaAplikasi(account);

    } else {

        alert(
            "Username, password, atau role salah!"
        );

    }

}


/* ==================================================
   BUKA APLIKASI
================================================== */

function bukaAplikasi(account) {

    document.getElementById(
        "loginPage"
    ).style.display = "none";


    document.getElementById(
        "app"
    ).style.display = "block";


    document.getElementById(
        "currentUser"
    ).textContent =
        account.username +
        " (" +
        account.role.toUpperCase() +
        ")";


    const attendanceNav =
        document.getElementById(
            "attendanceNav"
        );

    const reportNav =
        document.getElementById(
            "reportNav"
        );


    /*
       FORM ABSENSI HANYA UNTUK MURID
    */

    if (account.role === "murid") {

        attendanceNav.style.display =
            "block";

        reportNav.style.display =
            "none";

    }

    /*
       GURU DAN ADMIN
    */

    else {

        attendanceNav.style.display =
            "none";

        reportNav.style.display =
            "block";

    }


    updateDashboard();

    tampilkanData();

}


/* ==================================================
   CEK LOGIN SAAT REFRESH
================================================== */

window.onload = function() {

    const savedUser =
        localStorage.getItem(
            "loginUser"
        );


    if (savedUser) {

        bukaAplikasi(
            JSON.parse(savedUser)
        );

    }

};


/* ==================================================
   LOGOUT
================================================== */

function logout() {

    localStorage.removeItem(
        "loginUser"
    );

    location.reload();

}


/* ==================================================
   NAVIGASI
================================================== */

function showPage(pageId, button) {

    document.querySelectorAll(
        ".page"
    ).forEach(page => {

        page.classList.remove(
            "active"
        );

    });


    document.getElementById(
        pageId
    ).classList.add(
        "active"
    );


    document.querySelectorAll(
        ".nav button"
    ).forEach(btn => {

        btn.classList.remove(
            "active"
        );

    });


    button.classList.add(
        "active"
    );

}


/* ==================================================
   PREVIEW FOTO
================================================== */

function previewFoto(event) {

    const file =
        event.target.files[0];


    if (!file) return;


    /*
       MAKSIMAL 20 MB
    */

    const maxSize =
        20 * 1024 * 1024;


    if (file.size > maxSize) {

        alert(
            "Ukuran foto maksimal 20 MB!"
        );


        document.getElementById(
            "foto"
        ).value = "";


        fotoBase64 = "";


        document.getElementById(
            "photoPreview"
        ).innerHTML =
            "Foto belum dipilih";


        return;

    }


    const reader =
        new FileReader();


    reader.onload =
        function(e) {

            fotoBase64 =
                e.target.result;


            document.getElementById(
                "photoPreview"
            ).innerHTML =

                `<img
                    src="${fotoBase64}"
                    alt="Foto Kehadiran">`;

        };


    reader.readAsDataURL(file);

}
